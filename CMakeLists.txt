cmake_minimum_required(VERSION 3.21)
project(FIFA18SaveEditor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Concurrent)
find_package(LibArchive REQUIRED)
qt_standard_project_setup()
qt_add_executable(fifa18-save-editor WIN32 MACOSX_BUNDLE src/main.cpp src/career.cpp)
qt_add_resources(fifa18-save-editor schema PREFIX "/" FILES schema.json)
target_link_libraries(fifa18-save-editor PRIVATE Qt6::Core Qt6::Widgets Qt6::Concurrent LibArchive::LibArchive)
set_target_properties(fifa18-save-editor PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "FIFA 18 Save Editor" MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.fifa18saveeditor" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
include(GNUInstallDirs)
install(TARGETS fifa18-save-editor BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md FORMAT.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/fifa18-save-editor)
if(UNIX AND NOT APPLE)
  install(FILES fifa18-save-editor.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()
if(WIN32 AND DEFINED VCPKG_INSTALLED_DIR)
  install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin/" DESTINATION ${CMAKE_INSTALL_BINDIR} FILES_MATCHING PATTERN "*.dll")
endif()
if(APPLE OR WIN32)
  qt_generate_deploy_app_script(TARGET fifa18-save-editor OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
set(CPACK_PACKAGE_NAME "fifa18-save-editor")
set(CPACK_PACKAGE_CONTACT "FIFA18 Save Editor contributors")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
elseif(APPLE)
  set(CPACK_GENERATOR "DragNDrop")
else()
  set(CPACK_GENERATOR "DEB;RPM")
endif()
include(CPack)
